cmake_minimum_required(VERSION 3.20)
project(edikka_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ABSL_PROPAGATE_CXX_STD ON)
include(FetchContent)
FetchContent_Declare(abseil GIT_REPOSITORY https://github.com/abseil/abseil-cpp.git GIT_TAG d9e4955c65cd4367dd6bf46f4ccb8cd3d100540b)
FetchContent_MakeAvailable(abseil)
add_executable(robots-oracle main.cc ../vendor/google-robotstxt/robots.cc)
target_include_directories(robots-oracle PRIVATE ../vendor/google-robotstxt)
target_link_libraries(robots-oracle absl::base absl::strings absl::fixed_array)
